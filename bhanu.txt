Query 1: Display All Orders with Customer, Restaurant, and Menu Item Details


mysql> SELECT
    ->     c.customer_name,
    ->     r.restaurant_name,
    ->     m.item_name,
    ->     o.quantity,
    ->     o.total_amount
    -> FROM Orders o
    -> JOIN Customers c
    -> ON o.customer_id = c.customer_id
    -> JOIN Restaurants r
    -> ON o.restaurant_id = r.restaurant_id
    -> JOIN MenuItems m
    -> ON o.item_id = m.item_id;
